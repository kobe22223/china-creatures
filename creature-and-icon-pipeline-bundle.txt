"""Sculpting toolkit for studded creature meshes (bpy).

Write creatures in "game" coordinates: P(x, y, z) with y = up and z = backwards (the creature's
front points to -z), like Roblox. Internally Blender is Z-up with the front at -Y.

Every shape belongs to one or more bones (rigid or blended weights), has a colour, and flags:
  stud=False  -> no stud pattern (eyes, mouth, gems)
  glow=True   -> exported as a separate "Glow_*" mesh that Roblox renders as Neon
The final mesh is UV-unwrapped and gets a baked texture: flat colours + shading + studs.
"""
import math, json, os, random
import bpy, bmesh
import numpy as np
from mathutils import Vector, Matrix, Euler, Quaternion
from PIL import Image, ImageDraw, ImageFilter


# game (x right, y up, z back) -> blender: a proper rotation, matching the FBX export axes
# (blender -Y forward / Z up -> FBX -Z forward / Y up), so game x = -blender x.
G = Matrix(((-1, 0, 0), (0, 0, 1), (0, 1, 0)))


def P(x, y, z):
    return Vector((-x, z, y))


def C(r, g, b):
    return (r / 255, g / 255, b / 255)


def euler(rx=0, ry=0, rz=0):
    """rotation given in game axes (degrees), returned as a blender-space matrix"""
    return G @ Euler((math.radians(rx), math.radians(ry), math.radians(rz)), "XYZ").to_matrix() @ G


STUD_SCALE = 0.55


class Creature:
    def __init__(self, name, height=9.0, stud=None):
        self.name = name
        self.height = height
        self.stud = (stud or height / 30) * STUD_SCALE   # owner asked for denser studs
        self.bones = []          # (name, head, tail, parent)
        self.shapes = []         # dicts: bm, color, bones, glow, stud
        self.palette = []
        self.anim_spec = {}
        self.extra = {}

    # ---------------------------------------------------------------- skeleton
    def bn(self, n):
        """exported bone name: unique per creature (Roblox's importer merges same-named bones across a batch)"""
        return self.name + "__" + n

    def bone(self, name, head, tail, parent=None):
        self.bones.append((name, Vector(head), Vector(tail), parent))
        return name

    def bone_pos(self, name, end="head"):
        for n, h, t, p in self.bones:
            if n == name:
                return h if end == "head" else t
        raise KeyError(name)

    # ---------------------------------------------------------------- shapes
    def _add(self, bm, color, bones, glow=False, stud=True, smooth=False, paint=None):
        """paint = fn(x, y, z game coords of a face centre) -> colour or None (painted patterns: stripes, spots)"""
        self.shapes.append(dict(bm=bm, color=tuple(color), bones=bones if isinstance(bones, (list, tuple)) else [bones],
                                glow=glow, stud=stud and not glow, smooth=smooth, paint=paint))
        return bm

    def ellipsoid(self, center, radii, color, bones, rot=None, seg=(14, 9), **kw):
        bm = bmesh.new()
        bmesh.ops.create_uvsphere(bm, u_segments=seg[0], v_segments=seg[1], radius=1.0)
        m = Matrix.Diagonal((radii[0], radii[2], radii[1], 1.0))  # game (x,y,z) radii -> blender (x,z,y)
        if rot is not None:
            m = rot.to_4x4() @ m
        m = Matrix.Translation(center) @ m
        bmesh.ops.transform(bm, matrix=m, verts=bm.verts)
        return self._add(bm, color, bones, **kw)

    def tube(self, points, radii, color, bones, seg=10, cap=True, flat=None, sub=3, **kw):
        """tapered tube through points (Vectors) with per-point radius. flat = (sx, sy) squash of the
        cross-section (sx along the 'side' axis, sy along the 'up-ish' axis). sub = smooth subdivision."""
        if sub > 1 and len(points) >= 3:
            pts2, rad2 = [], []
            n0 = len(points)
            for i in range(n0 - 1):
                p0, p1, p2, p3 = points[max(i - 1, 0)], points[i], points[i + 1], points[min(i + 2, n0 - 1)]
                for k in range(sub):
                    t = k / sub
                    t2, t3 = t * t, t * t * t
                    pts2.append(0.5 * ((2 * p1) + (-p0 + p2) * t + (2 * p0 - 5 * p1 + 4 * p2 - p3) * t2 + (-p0 + 3 * p1 - 3 * p2 + p3) * t3))
                    rad2.append(radii[i] + (radii[i + 1] - radii[i]) * t)
            pts2.append(points[-1]); rad2.append(radii[-1])
            points, radii = pts2, rad2
        bm = bmesh.new()
        rings = []
        n = len(points)
        up_hint = Vector((0, 0, 1))
        for i, p in enumerate(points):
            d = (points[min(i + 1, n - 1)] - points[max(i - 1, 0)]).normalized()
            side = d.cross(up_hint)
            if side.length < 1e-3:
                side = d.cross(Vector((1, 0, 0)))
            side.normalize()
            up = side.cross(d).normalized()
            r = radii[i]
            sx, sy = flat or (1, 1)
            ring = []
            for k in range(seg):
                a = k / seg * math.tau
                ring.append(bm.verts.new(p + side * math.cos(a) * r * sx + up * math.sin(a) * r * sy))
            rings.append(ring)
        for i in range(n - 1):
            for k in range(seg):
                a, b = rings[i][k], rings[i][(k + 1) % seg]
                c, d2 = rings[i + 1][(k + 1) % seg], rings[i + 1][k]
                bm.faces.new((a, b, c, d2))
        if cap:
            for ring, pt, sign in ((rings[0], points[0], -1), (rings[-1], points[-1], 1)):
                dirv = (points[1] - points[0]) if sign < 0 else (points[-1] - points[-2])
                tip = bm.verts.new(pt + dirv.normalized() * radii[0 if sign < 0 else -1] * 0.35 * sign)
                for k in range(seg):
                    v1, v2 = ring[k], ring[(k + 1) % seg]
                    if sign < 0:
                        bm.faces.new((v2, v1, tip))
                    else:
                        bm.faces.new((v1, v2, tip))
        bmesh.ops.recalc_face_normals(bm, faces=bm.faces)
        return self._add(bm, color, bones, **kw)

    def curve_points(self, a, b, bend=Vector((0, 0, 0)), n=6):
        """quadratic curve from a to b, bulging by `bend` at the middle"""
        pts = []
        for i in range(n):
            t = i / (n - 1)
            pts.append(a.lerp(b, t) + bend * (4 * t * (1 - t)))
        return pts

    def spike(self, base, tip, r, color, bones, bend=None, seg=7, n=5, **kw):
        pts = self.curve_points(base, tip, bend or Vector((0, 0, 0)), n)
        radii = [r * (1 - i / (n - 1)) ** 1.1 + 0.02 for i in range(n)]
        return self.tube(pts, radii, color, bones, seg=seg, **kw)

    def leaf(self, origin, along, width_dir, length, width, color, bones, thick=0.12, n=7, shape="feather", curl=0.0, **kw):
        """flat shape (feather, fin, ear, flame, membrane) with thickness"""
        along = along.normalized()
        width_dir = (width_dir - along * width_dir.dot(along)).normalized()
        normal = along.cross(width_dir).normalized()
        top, bot = [], []
        prof = []
        for i in range(n):
            t = i / (n - 1)
            if shape == "feather":
                w = math.sin(math.pi * min(1, t * 1.15)) ** 0.7 * (1 - 0.35 * t)
            elif shape == "flame":
                w = math.sin(math.pi * t) ** 0.6 * (1 - t) ** 0.5 * 1.4
            elif shape == "fin":
                w = (1 - t) ** 0.8
            else:  # ear / blade
                w = math.sin(math.pi * (0.5 + 0.5 * t)) if t > 0 else 1
            prof.append(max(w, 0.02))
        bm = bmesh.new()
        for i, w in enumerate(prof):
            t = i / (n - 1)
            c = origin + along * (t * length) + normal * (curl * t * t * length)
            for s in (-1, 1):
                for h in (-1, 1):
                    pass
            top.append((bm.verts.new(c - width_dir * w * width / 2 + normal * thick / 2), bm.verts.new(c + width_dir * w * width / 2 + normal * thick / 2)))
            bot.append((bm.verts.new(c - width_dir * w * width / 2 - normal * thick / 2), bm.verts.new(c + width_dir * w * width / 2 - normal * thick / 2)))
        for i in range(n - 1):
            bm.faces.new((top[i][0], top[i][1], top[i + 1][1], top[i + 1][0]))
            bm.faces.new((bot[i][0], bot[i + 1][0], bot[i + 1][1], bot[i][1]))
            bm.faces.new((top[i][0], top[i + 1][0], bot[i + 1][0], bot[i][0]))
            bm.faces.new((top[i][1], bot[i][1], bot[i + 1][1], top[i + 1][1]))
        bm.faces.new((top[0][0], bot[0][0], bot[0][1], top[0][1]))
        bm.faces.new((top[-1][0], top[-1][1], bot[-1][1], bot[-1][0]))
        bmesh.ops.recalc_face_normals(bm, faces=bm.faces)
        return self._add(bm, color, bones, **kw)

    def sheet(self, pts, color, bones, thick=0.12, **kw):
        """thin solid panel through a loop of 3D points (fan from their average): wing membranes, fins"""
        bm = bmesh.new()
        ctr = sum(pts, Vector()) / len(pts)
        n = Vector()
        for i in range(len(pts)):
            n += (pts[i] - ctr).cross(pts[(i + 1) % len(pts)] - ctr)
        n = n.normalized() * thick / 2
        top = [bm.verts.new(p + n) for p in pts]; bot = [bm.verts.new(p - n) for p in pts]
        ct, cb = bm.verts.new(ctr + n), bm.verts.new(ctr - n)
        m = len(pts)
        for i in range(m):
            j = (i + 1) % m
            bm.faces.new((ct, top[i], top[j]))
            bm.faces.new((cb, bot[j], bot[i]))
            bm.faces.new((top[i], bot[i], bot[j], top[j]))
        bmesh.ops.recalc_face_normals(bm, faces=bm.faces)
        return self._add(bm, color, bones, **kw)

    def torus(self, center, R, r, color, bones, rot=None, seg=(24, 8), **kw):
        bm = bmesh.new()
        rings = []
        for i in range(seg[0]):
            a = i / seg[0] * math.tau
            ring = []
            for k in range(seg[1]):
                b = k / seg[1] * math.tau
                x = (R + r * math.cos(b)) * math.cos(a)
                z = (R + r * math.cos(b)) * math.sin(a)
                y = r * math.sin(b)
                ring.append(bm.verts.new(P(x, y, z)))
            rings.append(ring)
        for i in range(seg[0]):
            for k in range(seg[1]):
                a, b = rings[i][k], rings[i][(k + 1) % seg[1]]
                c, d = rings[(i + 1) % seg[0]][(k + 1) % seg[1]], rings[(i + 1) % seg[0]][k]
                bm.faces.new((a, d, c, b))
        m = Matrix.Translation(center) @ ((rot.to_4x4()) if rot else Matrix.Identity(4))
        bmesh.ops.transform(bm, matrix=m, verts=bm.verts)
        bmesh.ops.recalc_face_normals(bm, faces=bm.faces)
        return self._add(bm, color, bones, **kw)

    def box(self, center, half, color, bones, rot=None, **kw):
        bm = bmesh.new()
        bmesh.ops.create_cube(bm, size=2.0)
        m = Matrix.Diagonal((half[0], half[2], half[1], 1.0))
        if rot is not None:
            m = rot.to_4x4() @ m
        bmesh.ops.transform(bm, matrix=Matrix.Translation(center) @ m, verts=bm.verts)
        return self._add(bm, color, bones, **kw)

    # cartoon eye: white, coloured iris, black pupil, two highlights. `facing` = outward direction
    def snap(self, point, direction, max_dist=4.0):
        """move a point onto the outer surface of the shapes built so far, looking back along -direction"""
        from mathutils.bvhtree import BVHTree
        d = direction.normalized()
        start = point + d * max_dist
        best = None
        for sh in self.shapes:
            if sh["glow"]:
                continue
            tree = BVHTree.FromBMesh(sh["bm"])
            hit = tree.ray_cast(start, -d, max_dist * 2)
            if hit[0] is not None:
                dist = (hit[0] - start).length
                if best is None or dist < best[1]:
                    best = (hit[0], dist, hit[1])
        return (best[0], best[2]) if best else (point, d)

    def eye(self, center, size, facing, bones, iris=C(60, 140, 230), lid=None, angry=0.0, style="black", snap=True):
        if snap:
            center, nrm = self.snap(center, facing)
            facing = (facing.normalized() * 0.5 + nrm.normalized() * 0.5)
        f = facing.normalized()
        up = Vector((0, 0, 1))
        side = f.cross(up).normalized()
        upv = side.cross(f).normalized()
        rot = Matrix((side, f, upv)).transposed()  # columns: x=side, y=facing, z=up (blender)
        def ell(off, rad, col):
            bm = bmesh.new()
            bmesh.ops.create_uvsphere(bm, u_segments=16, v_segments=10, radius=1.0)
            m = Matrix.Translation(center + side * off[0] + f * off[1] + upv * off[2]) @ rot.to_4x4() @ Matrix.Diagonal((rad[0], rad[1], rad[2], 1))
            bmesh.ops.transform(bm, matrix=m, verts=bm.verts)
            self._add(bm, col, bones, stud=False, smooth=True)
        s = size
        if style == "black":
            # flat black eye: a thin oval pushed into the surface so only a flush, flat face shows
            ell((0, -s * 0.12, 0), (s * 0.62, s * 0.2, s * 0.85), C(12, 12, 16))
        else:
            ell((0, 0, 0), (s * 0.85, s * 0.45, s), C(255, 255, 255))
            ell((0, s * 0.18, -s * 0.08), (s * 0.6, s * 0.35, s * 0.68), iris)
            ell((0, s * 0.3, -s * 0.1), (s * 0.36, s * 0.28, s * 0.42), C(15, 15, 25))
            ell((-s * 0.22, s * 0.48, s * 0.25), (s * 0.17, s * 0.1, s * 0.17), C(255, 255, 255))
            ell((s * 0.2, s * 0.44, -s * 0.32), (s * 0.08, s * 0.06, s * 0.08), C(255, 255, 255))
        if lid:
            bm = bmesh.new()
            bmesh.ops.create_uvsphere(bm, u_segments=16, v_segments=10, radius=1.0)
            m = Matrix.Translation(center + upv * s * 0.55 + f * s * 0.05) @ rot.to_4x4() @ Matrix.Rotation(angry, 4, "Y") @ Matrix.Diagonal((s * 1.0, s * 0.5, s * 0.35, 1))
            bmesh.ops.transform(bm, matrix=m, verts=bm.verts)
            self._add(bm, lid, bones)

    # ---------------------------------------------------------------- build
    def _weights(self, verts, bones):
        """weights for a shape: single bone -> 1.0; several -> blend by distance to bone segments"""
        if len(bones) == 1:
            return [{bones[0]: 1.0} for _ in verts]
        segs = []
        for b in bones:
            for n, h, t, p in self.bones:
                if n == b:
                    segs.append((b, h, t))
        out = []
        for v in verts:
            ds = []
            for b, h, t in segs:
                ab = t - h
                tt = max(0, min(1, (v - h).dot(ab) / ab.length_squared))
                ds.append((b, (v - (h + ab * tt)).length + 1e-4))
            ws = [(b, 1 / d ** 4) for b, d in ds]
            tot = sum(w for _, w in ws)
            ws = sorted(((b, w / tot) for b, w in ws), key=lambda x: -x[1])[:2]
            tot = sum(w for _, w in ws)
            out.append({b: w / tot for b, w in ws})
        return out

    def build(self, outdir):
        os.makedirs(outdir, exist_ok=True)
        bpy.ops.wm.read_factory_settings(use_empty=True)
        scn = bpy.context.scene
        # armature
        arm_data = bpy.data.armatures.new(self.name + "_Rig")
        arm = bpy.data.objects.new(self.name, arm_data)
        scn.collection.objects.link(arm)
        bpy.context.view_layer.objects.active = arm
        bpy.ops.object.mode_set(mode="EDIT")
        for n, h, t, p in self.bones:
            eb = arm_data.edit_bones.new(self.bn(n))
            eb.head, eb.tail = h, t
            eb.roll = 0
        for n, h, t, p in self.bones:
            if p:
                eb = arm_data.edit_bones[self.bn(n)]
                eb.parent = arm_data.edit_bones[self.bn(p)]
                eb.use_connect = False
        bpy.ops.object.mode_set(mode="OBJECT")
        self.arm = arm
        # palette
        cols = []
        for s in self.shapes:
            if s["color"] not in cols:
                cols.append(s["color"])
        self.palette = cols
        self.paints = []
        # meshes: one "Body" mesh (studded, textured) + one mesh per glow colour
        groups = {"Body": [s for s in self.shapes if not s["glow"]]}
        for s in self.shapes:
            if s["glow"]:
                key = "Glow_%02x%02x%02x" % tuple(int(c * 255) for c in s["color"])
                groups.setdefault(key, []).append(s)
        self.mesh_objs = {}
        for gname, shapes in groups.items():
            bm = bmesh.new()
            col_layer = bm.faces.layers.int.new("col")
            flag_layer = bm.faces.layers.int.new("flag")
            paint_layer = bm.faces.layers.int.new("paint")
            deform = bm.verts.layers.deform.verify()
            bone_index = {n: i for i, (n, *_r) in enumerate(self.bones)}
            for s in shapes:
                src = s["bm"]
                vmap = {}
                ws = self._weights([v.co for v in src.verts], s["bones"])
                for v, w in zip(src.verts, ws):
                    nv = bm.verts.new(v.co)
                    for b, wt in w.items():
                        nv[deform][bone_index[b]] = wt
                    vmap[v] = nv
                ci = cols.index(s["color"])
                pidx = 0
                if s.get("paint"):
                    self.paints.append(s["paint"])
                    pidx = len(self.paints)
                for f in src.faces:
                    try:
                        nf = bm.faces.new([vmap[v] for v in f.verts])
                    except ValueError:
                        continue
                    nf[col_layer] = ci
                    nf[paint_layer] = pidx
                    nf[flag_layer] = (1 if s["stud"] else 0) | (2 if s["smooth"] else 0)
                    nf.smooth = s["smooth"]
            uname = self.name + "_" + gname          # unique per creature (batch FBX files hold several)
            me = bpy.data.meshes.new(uname)
            bm.to_mesh(me)
            bm.free()
            ob = bpy.data.objects.new(uname, me)
            scn.collection.objects.link(ob)
            for n, *_r in self.bones:
                ob.vertex_groups.new(name=self.bn(n))
            ob.parent = arm
            mod = ob.modifiers.new("Armature", "ARMATURE")
            mod.object = arm
            self.mesh_objs[gname] = ob
        self._texture(outdir)
        return self

    # ---------------------------------------------------------------- texture
    def _texture(self, outdir, size=2048):
        body = self.mesh_objs["Body"]
        bpy.ops.object.select_all(action="DESELECT")
        body.select_set(True)
        bpy.context.view_layer.objects.active = body
        bpy.ops.object.mode_set(mode="EDIT")
        bpy.ops.mesh.select_all(action="SELECT")
        # lightmap pack: every face tiled tightly (smart project used only ~10% of the texture on faceted fur)
        bpy.ops.uv.lightmap_pack(PREF_CONTEXT="ALL_FACES", PREF_PACK_IN_ONE=True, PREF_BOX_DIV=48, PREF_MARGIN_DIV=0.15)
        bpy.ops.object.mode_set(mode="OBJECT")
        me = body.data
        uv = me.uv_layers.active.data
        cols = me.attributes["col"].data
        flags = me.attributes["flag"].data
        img = Image.new("RGB", (size, size), (0, 0, 0))
        mask = Image.new("L", (size, size), 0)       # 255 = has stud pattern
        filled = Image.new("L", (size, size), 0)
        dr, dm, df = ImageDraw.Draw(img), ImageDraw.Draw(mask), ImageDraw.Draw(filled)
        rng = random.Random(7)
        world_area, uv_area = 0.0, 0.0
        for poly in me.polygons:
            pts = [uv[li].uv for li in poly.loop_indices]
            a = 0.0
            for i in range(len(pts)):
                x1, y1 = pts[i]; x2, y2 = pts[(i + 1) % len(pts)]
                a += x1 * y2 - x2 * y1
            uv_area += abs(a) / 2
            world_area += poly.area
            base = self.palette[cols[poly.index].value]
            nz = poly.normal.z
            shade = 1.0 + 0.10 * nz - (0.10 if nz < -0.3 else 0) + rng.uniform(-0.025, 0.025)
            if flags[poly.index].value & 2:
                shade = 1.0
            c = tuple(int(max(0, min(255, ch * 255 * shade))) for ch in base)
            xy = [(p[0] * size, (1 - p[1]) * size) for p in pts]
            dr.polygon(xy, fill=c, outline=c)
            df.polygon(xy, fill=255, outline=255)
            if flags[poly.index].value & 1:
                dm.polygon(xy, fill=255, outline=255)
        # bleed colours into empty pixels so UV seams don't show black lines
        arr = np.array(img).astype(np.int32)
        fm = np.array(filled) > 0
        for _ in range(24):
            grown = arr.copy(); gm = fm.copy()
            for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                sh = np.roll(np.roll(arr, dx, 1), dy, 0)
                shm = np.roll(np.roll(fm, dx, 1), dy, 0)
                take = shm & ~gm
                grown[take] = sh[take]; gm |= take
            arr, fm = grown, gm
        # per-pixel painted patterns (stripes, spots...): every texel finds its 3D spot on the body
        if self.paints:
            pa = me.attributes["paint"].data
            co = np.array([(-v.co.x, v.co.z, v.co.y) for v in me.vertices])   # game coords
            for poly in me.polygons:
                pi = pa[poly.index].value
                if pi == 0:
                    continue
                col, maskfn = self.paints[pi - 1]
                lis = list(poly.loop_indices)
                for tri in range(1, len(lis) - 1):
                    ids = (lis[0], lis[tri], lis[tri + 1])
                    uvp = np.array([(uv[i].uv[0] * size, (1 - uv[i].uv[1]) * size) for i in ids])
                    p3 = co[[me.loops[i].vertex_index for i in ids]]
                    x0, y0 = np.floor(uvp.min(0)).astype(int) - 1
                    x1, y1 = np.ceil(uvp.max(0)).astype(int) + 1
                    x0, y0 = max(x0, 0), max(y0, 0); x1, y1 = min(x1, size - 1), min(y1, size - 1)
                    if x1 <= x0 or y1 <= y0:
                        continue
                    gy, gx = np.mgrid[y0:y1 + 1, x0:x1 + 1]
                    (ax, ay), (bx, by), (cx, cy) = uvp
                    den = (by - cy) * (ax - cx) + (cx - bx) * (ay - cy)
                    if abs(den) < 1e-9:
                        continue
                    l1 = ((by - cy) * (gx + 0.5 - cx) + (cx - bx) * (gy + 0.5 - cy)) / den
                    l2 = ((cy - ay) * (gx + 0.5 - cx) + (ax - cx) * (gy + 0.5 - cy)) / den
                    l3 = 1 - l1 - l2
                    inside = (l1 >= -0.02) & (l2 >= -0.02) & (l3 >= -0.02)
                    if not inside.any():
                        continue
                    X = l1 * p3[0, 0] + l2 * p3[1, 0] + l3 * p3[2, 0]
                    Y = l1 * p3[0, 1] + l2 * p3[1, 1] + l3 * p3[2, 1]
                    Z = l1 * p3[0, 2] + l2 * p3[1, 2] + l3 * p3[2, 2]
                    m2 = inside & maskfn(X, Y, Z)
                    if m2.any():
                        shade = arr[gy[m2], gx[m2]].astype(np.float32) / (np.array(self.palette[cols[poly.index].value]) * 255 + 1e-3)
                        shade = np.clip(shade.mean(-1, keepdims=True), 0.7, 1.3)
                        arr[gy[m2], gx[m2]] = np.clip(np.array(col) * 255 * shade, 0, 255).astype(np.int32)
        # stud pattern evaluated in 3D (world units) per texel, so every stud is the same size whatever the UV
        # layout does. Each face projects onto the two axes most parallel to it.
        ppu = math.sqrt(uv_area * size * size / max(world_area, 1e-6))
        print("STUDTILE", round(self.stud * ppu, 1), "px", "cover", round(uv_area, 3))
        studs = np.ones((size, size), np.float32)
        co = np.array([(-v.co.x, v.co.z, v.co.y) for v in me.vertices])   # game coords
        m = 0.16
        k_ = getattr(self, "stud_k", 1.0)

        def pattern(Ua, Va):
            u = Ua - np.floor(Ua); v = Va - np.floor(Va)
            inner = (u > m) & (u < 1 - m) & (v > m) & (v < 1 - m)
            cu, cv = (u - 0.5) / (0.5 - m), (v - 0.5) / (0.5 - m)
            mx = np.maximum(np.abs(cu), np.abs(cv))
            edge = inner & (mx > 0.72)
            out = np.ones_like(u)
            out[inner] = 1.03
            out[edge & (((cu < 0) & (np.abs(cu) >= np.abs(cv))) | ((cv > 0) & (np.abs(cv) >= np.abs(cu))))] = 1.14
            out[edge & (((cu > 0) & (np.abs(cu) >= np.abs(cv))) | ((cv < 0) & (np.abs(cv) >= np.abs(cu))))] = 0.8
            out[(u < 0.05) | (v < 0.05)] = 0.94
            return 1 + (out - 1) * k_
        for poly in me.polygons:
            if not (flags[poly.index].value & 1):
                continue
            nb = poly.normal
            ng = (abs(nb.x), abs(nb.z), abs(nb.y))          # game |x|,|y|,|z|
            ax = (2, 1) if ng[0] >= max(ng[1], ng[2]) else ((0, 2) if ng[1] >= ng[2] else (0, 1))
            lis = list(poly.loop_indices)
            for tri in range(1, len(lis) - 1):
                ids = (lis[0], lis[tri], lis[tri + 1])
                uvp = np.array([(uv[i].uv[0] * size, (1 - uv[i].uv[1]) * size) for i in ids])
                p3 = co[[me.loops[i].vertex_index for i in ids]]
                x0, y0 = np.floor(uvp.min(0)).astype(int) - 1
                x1, y1 = np.ceil(uvp.max(0)).astype(int) + 1
                x0, y0 = max(x0, 0), max(y0, 0); x1, y1 = min(x1, size - 1), min(y1, size - 1)
                if x1 <= x0 or y1 <= y0:
                    continue
                gy, gx = np.mgrid[y0:y1 + 1, x0:x1 + 1]
                (ax_, ay_), (bx_, by_), (cx_, cy_) = uvp
                den = (by_ - cy_) * (ax_ - cx_) + (cx_ - bx_) * (ay_ - cy_)
                if abs(den) < 1e-9:
                    continue
                l1 = ((by_ - cy_) * (gx + 0.5 - cx_) + (cx_ - bx_) * (gy + 0.5 - cy_)) / den
                l2 = ((cy_ - ay_) * (gx + 0.5 - cx_) + (ax_ - cx_) * (gy + 0.5 - cy_)) / den
                l3 = 1 - l1 - l2
                inside = (l1 >= -0.08) & (l2 >= -0.08) & (l3 >= -0.08)
                if not inside.any():
                    continue
                A = l1 * p3[0, ax[0]] + l2 * p3[1, ax[0]] + l3 * p3[2, ax[0]]
                Bv = l1 * p3[0, ax[1]] + l2 * p3[1, ax[1]] + l3 * p3[2, ax[1]]
                pat = pattern(A[inside] / self.stud, Bv[inside] / self.stud)
                studs[gy[inside], gx[inside]] = pat
        f = studs[..., None]
        out = np.clip(arr * f, 0, 255).astype(np.uint8)
        self.tex_path = os.path.join(outdir, self.name + "_Tex.png")
        Image.fromarray(out).save(self.tex_path)
        # material
        mat = bpy.data.materials.new(self.name + "_Mat")
        mat.use_nodes = True
        nt = mat.node_tree
        bsdf = nt.nodes["Principled BSDF"]
        bsdf.inputs["Roughness"].default_value = 0.6
        tn = nt.nodes.new("ShaderNodeTexImage")
        tn.image = bpy.data.images.load(self.tex_path)
        nt.links.new(tn.outputs["Color"], bsdf.inputs["Base Color"])
        body.data.materials.append(mat)
        for gname, ob in self.mesh_objs.items():
            if gname.startswith("Glow_"):
                gm = bpy.data.materials.new(gname + "_Mat")
                gm.use_nodes = True
                b = gm.node_tree.nodes["Principled BSDF"]
                hexs = gname[5:]
                rgbv = tuple(int(hexs[i:i + 2], 16) / 255 for i in (0, 2, 4))
                b.inputs["Base Color"].default_value = rgbv + (1,)
                b.inputs["Emission Color"].default_value = rgbv + (1,)
                b.inputs["Emission Strength"].default_value = 1.4
                ob.data.materials.append(gm)

    # ---------------------------------------------------------------- animation
    def animate(self, clips):
        """clips = {"Idle": (frames, seconds, fn), ...} where fn(t01) returns
        {bone: (rx, ry, rz) degrees in game axes (local, applied around the bone head) or (rot, (dx,dy,dz))}"""
        arm = self.arm
        bpy.context.view_layer.objects.active = arm
        bpy.ops.object.mode_set(mode="POSE")
        self.anim_data = {}
        rest = {b.name: b.matrix_local.copy() for b in arm.data.bones}
        rest = {n: rest[self.bn(n)] for n, *_r in self.bones}
        for clip, (frames, seconds, fn) in clips.items():
            data = []
            for fi in range(frames):
                t = fi / frames
                pose = fn(t)
                # reset
                for pb in arm.pose.bones:
                    pb.matrix_basis = Matrix.Identity(4)
                bpy.context.view_layer.update()
                # apply top-down: rotation around the bone head in WORLD(armature) axes, given in game axes
                order = [n for n, *_r in self.bones]
                for name in order:
                    if name not in pose:
                        continue
                    val = pose[name]
                    rot, off = (val, (0, 0, 0)) if len(val) == 3 and not isinstance(val[0], tuple) else val
                    pb = arm.pose.bones[self.bn(name)]
                    R = euler(*rot).to_4x4()   # game-axis rotation -> blender
                    cur = pb.matrix.copy()                          # armature-space current matrix
                    head = cur.to_translation()
                    M = Matrix.Translation(head + P(*off)) @ R @ Matrix.Translation(-head) @ cur
                    pb.matrix = M
                    bpy.context.view_layer.update()
                frame = {}
                for name in order:
                    pb = arm.pose.bones[self.bn(name)]
                    delta = pb.matrix @ rest[name].inverted()       # armature-space delta (rest -> posed)
                    # convert to game axes: Dg = G^-1 * delta * G
                    Dg = G.to_4x4() @ delta @ G.to_4x4()
                    q = Dg.to_quaternion()
                    tr = Dg.to_translation()
                    frame[name] = [round(q.w, 4), round(q.x, 4), round(q.y, 4), round(q.z, 4), round(tr.x, 3), round(tr.y, 3), round(tr.z, 3)]
                data.append(frame)
            self.anim_data[clip] = {"fps": round(frames / seconds, 3), "frames": data}
        for pb in arm.pose.bones:
            pb.matrix_basis = Matrix.Identity(4)
        bpy.ops.object.mode_set(mode="OBJECT")

    def pose_at(self, clip, fi):
        """apply a recorded frame to the blender rig (for preview renders)"""
        arm = self.arm
        bpy.context.view_layer.objects.active = arm
        bpy.ops.object.mode_set(mode="POSE")
        rest = {b.name: b.matrix_local.copy() for b in arm.data.bones}
        rest = {n: rest[self.bn(n)] for n, *_r in self.bones}
        for name, *_r in self.bones:
            w, x, y, z, tx, ty, tz = self.anim_data[clip]["frames"][fi][name]
            Dg = Matrix.Translation((tx, ty, tz)) @ Quaternion((w, x, y, z)).to_matrix().to_4x4()
            delta = G.to_4x4() @ Dg @ G.to_4x4()
            arm.pose.bones[self.bn(name)].matrix = delta @ rest[name]
            bpy.context.view_layer.update()
        bpy.ops.object.mode_set(mode="OBJECT")

    # ---------------------------------------------------------------- output
    def render(self, path, angle=30, elev=15, res=800, samples=20, dist=1.35):
        scn = bpy.context.scene
        scn.render.engine = "CYCLES"; scn.cycles.device = "CPU"; scn.cycles.samples = samples
        scn.cycles.use_denoising = True
        scn.render.resolution_x = scn.render.resolution_y = res
        if not scn.world:
            scn.world = bpy.data.worlds.new("W")
        scn.world.use_nodes = True
        bg = scn.world.node_tree.nodes["Background"]
        bg.inputs["Color"].default_value = (0.6, 0.78, 0.95, 1); bg.inputs["Strength"].default_value = 0.9
        mn = Vector((1e9,) * 3); mx = Vector((-1e9,) * 3)
        dg = bpy.context.evaluated_depsgraph_get()
        for ob in self.mesh_objs.values():
            ev = ob.evaluated_get(dg)
            for v in ev.to_mesh().vertices:
                w = ob.matrix_world @ v.co
                mn = Vector(map(min, mn, w)); mx = Vector(map(max, mx, w))
            ev.to_mesh_clear()
        ctr = (mn + mx) / 2; size = (mx - mn).length
        if "Sun" not in bpy.data.objects:
            sun = bpy.data.lights.new("Sun", "SUN"); sun.energy = 2.6; sun.angle = math.radians(10)
            so = bpy.data.objects.new("Sun", sun); scn.collection.objects.link(so)
            so.rotation_euler = (math.radians(45), math.radians(15), math.radians(-30))
            bpy.ops.mesh.primitive_plane_add(size=300, location=(ctr.x, ctr.y, 0))
            g = bpy.context.active_object; g.name = "Ground"
            gmat = bpy.data.materials.new("Gr"); gmat.use_nodes = True
            gmat.node_tree.nodes["Principled BSDF"].inputs["Base Color"].default_value = (0.3, 0.65, 0.22, 1)
            g.data.materials.append(gmat)
        cam = bpy.data.objects.get("Cam")
        if not cam:
            cd = bpy.data.cameras.new("Cam"); cd.lens = 50
            cam = bpy.data.objects.new("Cam", cd); scn.collection.objects.link(cam)
        scn.camera = cam
        a, e = math.radians(angle), math.radians(elev)
        d = size * dist
        cam.location = ctr + Vector((math.sin(a) * math.cos(e) * d, -math.cos(a) * math.cos(e) * d, math.sin(e) * d))
        cam.rotation_euler = (ctr - cam.location).to_track_quat("-Z", "Y").to_euler()
        scn.render.filepath = path
        bpy.ops.render.render(write_still=True)

    def export(self, outdir, scale_info=True):
        fbx = os.path.join(outdir, self.name + ".fbx")
        bpy.ops.object.select_all(action="DESELECT")
        self.arm.select_set(True)
        for ob in self.mesh_objs.values():
            ob.select_set(True)
        bpy.ops.export_scene.fbx(filepath=fbx, use_selection=True, object_types={"ARMATURE", "MESH"},
                                 path_mode="COPY", embed_textures=True, axis_forward="-Z", axis_up="Y",
                                 add_leaf_bones=False, bake_anim=False, use_armature_deform_only=False,
                                 mesh_smooth_type="FACE", apply_scale_options="FBX_SCALE_ALL")
        info = {"name": self.name, "height": self.height,
                "bones": [{"name": n, "parent": p, "head": [round(c, 3) for c in (-h.x, h.z, h.y)]} for n, h, t, p in self.bones],
                "glow": {self.name + "_" + g: g[5:] for g in self.mesh_objs if g.startswith("Glow_")},
                "body": self.name + "_Body",
                "anims": getattr(self, "anim_data", {}), "extra": self.extra}
        with open(os.path.join(outdir, self.name + "_data.json"), "w") as f:
            json.dump(info, f, separators=(",", ":"))
        return fbx

def quad(c, col, belly=None, L=1.0, H=1.0, W=1.0, leg=1.0, paw=None, chest_y=None, tail=3, neck=True, paint=None, chunk=1.0):

        c.bone("Shin" + S, P(s * x, sh_y * 0.55, hip_z - 0.6), P(s * x, 0.45, hip_z + 0.1), "Thigh" + S)
    if tail:
        t0 = P(0, sh_y + 0.2, hip_z + 1.1 * L)
        prev = "Hips"
        for i in range(tail):
            t1 = t0 + P(0, 0.35 * H, 1.15 * L)
            c.bone("Tail%d" % (i + 1), t0, t1, prev)
            prev, t0 = "Tail%d" % (i + 1), t1
    belly = belly or col
    r = 1.45 * min(H, W) ** 0.5 * (1 + 0.3 * (chunk - 1))
    c.tube([P(0, sh_y - 0.05, hip_z + 0.95 * L), P(0, sh_y, hip_z + 0.3 * L), P(0, sh_y + 0.1, 0.9 * L * 0.5), P(0, sh_y + 0.2, -0.5 * L), P(0, sh_y + 0.35 * H, chest_z + 0.4 * L), P(0, sh_y + 0.4 * H, chest_z - 0.35 * L)],
           [0.9 * r / 1.45, r * 0.95, r, r * 1.03, r * 1.1, r * 0.92], col, ["Hips", "Chest"], seg=18, flat=(W ** 0.5 * 0.92, 1.0), sub=5, paint=paint)
    c.ellipsoid(P(0, sh_y - r * 0.6, -0.2 * L), (r * 0.72 * W, r * 0.38, 2.1 * L), belly, ["Hips", "Chest"])

    pawc = paw or belly
    if chunk >= 1.3:
        # chunky legs: one smooth rounded column per leg (no flare, no seams) + round paws

            lr = 0.55 * leg * W ** 0.5 * chunk
            c.ellipsoid(P(s * x * 0.97, sh_y * 0.86, chest_z + 0.05), (lr * 1.18, sh_y * 0.28, lr * 1.3), col, "Arm" + S, seg=(20, 12), paint=paint)   # shoulder
            c.tube([P(s * x, sh_y * 0.86, chest_z + 0.05), P(s * x * 1.02, sh_y * 0.7, chest_z - 0.02), P(s * x * 1.03, sh_y * 0.5, chest_z - 0.1)],

            c.tube([P(s * x, sh_y * 0.7, hip_z - 0.2), P(s * x * 1.02, sh_y * 0.48, hip_z - 0.25), P(s * x * 1.04, sh_y * 0.28, hip_z - 0.05), P(s * x * 1.05, 0.55, hip_z + 0.1)],
                   [lr * 1.0, lr * 0.93, lr * 0.88, lr * 0.9], col, ["Shin" + S], seg=14, paint=paint)
            c.ellipsoid(P(s * x * 1.05, 0.45, hip_z - 0.12), (lr * 1.0, 0.45, lr * 1.12), pawc, "Shin" + S, seg=(18, 10))
            for t in (-1, 0, 1):   # toe beans
                for zz, bone in ((chest_z - 0.22 - lr * 1.05, "Fore" + S), (hip_z - 0.12 - lr * 1.02, "Shin" + S)):
                    c.ellipsoid(P(s * x * 1.05 + t * lr * 0.42, 0.32, zz + 0.12), (lr * 0.22, 0.18, lr * 0.16), pawc, bone, seg=(8, 6))
        return dict(sh_y=sh_y, hip_z=hip_z, chest_z=chest_z, x=x, head=hb, r=r)
    for s, S in ((-1, "L"), (1, "R")):
        lr = 0.55 * leg * W ** 0.5 * chunk
        tp = 1 + 0.5 * (chunk - 1)          # chunky legs taper less (thick all the way down)
        top = 1.45 if chunk <= 1.0 else 1.18          # mild chunk: thicker legs without the flared tops
        c.tube([P(s * x, sh_y + 0.2, chest_z + 0.15), P(s * x * 1.03, sh_y * 0.75, chest_z - 0.05), P(s * x * 1.03, sh_y * 0.5, chest_z - 0.12)], [lr * top, lr * 1.08, lr * 0.9], col, ["Arm" + S], seg=10, paint=paint)

        c.ellipsoid(P(s * x * 1.05, 0.33, hip_z - 0.1), (lr * 0.95 * tp, 0.32 * tp, lr * 1.2 * tp), pawc, "Shin" + S)
    return dict(sh_y=sh_y, hip_z=hip_z, chest_z=chest_z, x=x, head=hb, r=r)

def quad_anims(c, wings=False, tail=3, stride=24, bob=0.1, head_amp=1.0, extra=None):
    def walk(t):
        p = tau * t
        out = {
            "Root": ((0, 0, 0), (0, bob * abs(sin(2 * p)), 0)),
            "Chest": (1.5 * sin(2 * p), 2 * sin(p), 0),
            "Neck": (-3 * sin(2 * p) * head_amp, -2 * sin(p), 0),
            "Head": (2 * sin(2 * p + 0.5) * head_amp, 0, 0),
            "ArmL": (stride * sin(p), 0, 0), "ArmR": (-stride * sin(p), 0, 0),
            "ThighL": (-stride * sin(p), 0, 0), "ThighR": (stride * sin(p), 0, 0),
            "ForeL": (-28 * max(0, -sin(p - 0.6)), 0, 0), "ForeR": (-28 * max(0, sin(p - 0.6)), 0, 0),
            "ShinL": (24 * max(0, sin(p + 0.4)), 0, 0), "ShinR": (24 * max(0, -sin(p + 0.4)), 0, 0),
        }
        for i in range(tail):
            out["Tail%d" % (i + 1)] = (0, (12 + 2 * i) * sin(p - 0.7 * i), 0)
        if wings:
            out.update({"WingL": (0, 0, 8 + 7 * sin(2 * p)), "WingR": (0, 0, -8 - 7 * sin(2 * p)),
                        "WingTipL": (0, 0, 6 * sin(2 * p - 0.6)), "WingTipR": (0, 0, -6 * sin(2 * p - 0.6))})
        if extra:
            out.update(extra(t, True))
        return out

    def idle(t):
        p = tau * t
        out = {
            "Root": ((0, 0, 0), (0, 0.06 * sin(2 * p), 0)),
            "Chest": (1.5 * sin(2 * p), 0, 0),
            "Neck": (2 * sin(p), 14 * sin(p) * head_amp, 0),
            "Head": (4 * sin(2 * p + 1), 0, 4 * sin(p)),
        }
        for i in range(tail):
            out["Tail%d" % (i + 1)] = (0, (9 + 2 * i) * sin(p - 0.8 * i), 0)
        if wings:
            out.update({"WingL": (0, 0, 4 + 10 * sin(p)), "WingR": (0, 0, -4 - 10 * sin(p)),
                        "WingTipL": (0, 0, 8 * sin(p - 0.8)), "WingTipR": (0, 0, -8 * sin(p - 0.8))})
        if extra:
            out.update(extra(t, False))
        return out
    c.animate({"Walk": (24, 1.0, walk), "Idle": (40, 4.0, idle)})

def wing_bones(c, root, mid, tip, parent="Chest"):

def feather_wings(c, root, mid, tip, col, tipcol, cover=None, n=12, length=(2.4, 2.0), width=1.25, arm_col=None):
    for s, S in ((-1, "L"), (1, "R")):
        r, m, t = (P(s * root[0], root[1], root[2]), P(s * mid[0], mid[1], mid[2]), P(s * tip[0], tip[1], tip[2]))
        c.tube([r, m], [0.38, 0.3], arm_col or col, ["Wing" + S], seg=8)
        c.tube([m, t], [0.3, 0.16], arm_col or col, ["WingTip" + S], seg=8)
        for i in range(n):
            k = (i + 0.5) / n
            base = r.lerp(m, k * 2) if k < 0.5 else m.lerp(t, (k - 0.5) * 2)
            bone = "Wing" + S if k < 0.5 else "WingTip" + S
            ln = length[0] + length[1] * k
            down = P(s * 0.15, -1.0, 0.75 - 0.3 * k).normalized()
            arm_dir = (m - r) if k < 0.5 else (t - m)
            c.leaf(base, down, arm_dir, ln, width, col, bone, thick=0.16 + 0.02 * (i % 2), shape="feather", curl=0.05)
            c.leaf(base + down * (ln * 0.72) + P(0, 0, 0.02), down, arm_dir, ln * 0.32, width * 0.76, tipcol if i % 2 else tipcol, bone, thick=0.18, shape="feather")
        for i in range(5):
            k = (i + 0.5) / 5
            c.leaf(r.lerp(m, k) + P(0, 0.1, -0.12), P(s * 0.1, -1, 0.4), m - r, 1.4, 1.2, cover or col, "Wing" + S, thick=0.2, shape="feather")


def bat_wings(c, root, mid, tip, bone_col, skin_col, edge_col=None, fingers=4, glow_edge=False):
    """membrane wings: arm + finger bones with skin panels between them"""
    for s, S in ((-1, "L"), (1, "R")):
        r, m, t = (P(s * root[0], root[1], root[2]), P(s * mid[0], mid[1], mid[2]), P(s * tip[0], tip[1], tip[2]))
        c.tube([r, m], [0.34, 0.26], bone_col, ["Wing" + S], seg=8)
        c.tube([m, t], [0.26, 0.12], bone_col, ["WingTip" + S], seg=8)
        c.spike(m, m + P(s * 0.2, 0.7, -0.3), 0.18, edge_col or bone_col, "WingTip" + S, seg=5, n=3)   # thumb claw
        ends = []
        for f in range(fingers):
            k = f / max(1, fingers - 1)
            end = m + (P(s * (1.2 + 2.6 * (1 - k)), -1.0 - 3.2 * k, 0.6 + 2.4 * k))
            c.tube([m, m.lerp(end, 0.5), end], [0.16, 0.12, 0.06], bone_col, ["WingTip" + S], seg=6)
            ends.append(end)
        # membrane: one panel per gap between bones, scalloped trailing edge
        body = r + P(0, -1.8, 1.8)
        chain = [t] + ends + [body]
        for i in range(len(chain) - 1):
            a1, b1 = chain[i], chain[i + 1]
            mid_edge = (a1 + b1) / 2 + (m - (a1 + b1) / 2) * 0.22     # scallop pulls in toward the wrist
            if i == len(chain) - 2:
                panel = [m, a1, mid_edge, b1, r]
                bones = "Wing" + S
            else:
                panel = [m, a1, mid_edge, b1]
                bones = "WingTip" + S
            c.sheet(panel, skin_col, bones, thick=0.1)
        if edge_col and glow_edge:
            for e in ends:
                c.ellipsoid(e, (0.16, 0.16, 0.16), edge_col, "WingTip" + S, glow=True)

def biped(c, col, H=1.0, W=1.0, arm_len=1.0, leg_len=1.0, torso=None):
    """2-legged body (golems, warlocks, owls). Returns key points."""
    hip_y = 2.8 * leg_len * H
    sh_y = hip_y + 3.0 * H
    x_leg, x_arm = 0.75 * W, 1.9 * W
    c.bone("Root", P(0, 0, 0), P(0, 1, 0))
    c.bone("Hips", P(0, hip_y, 0), P(0, hip_y + 1.2 * H, 0), "Root")
    c.bone("Chest", P(0, hip_y + 1.2 * H, 0), P(0, sh_y, 0), "Hips")
    c.bone("Neck", P(0, sh_y, 0), P(0, sh_y + 0.6 * H, 0), "Chest")
    c.bone("Head", P(0, sh_y + 0.6 * H, 0), P(0, sh_y + 2.2 * H, 0), "Neck")
    for s, S in ((-1, "L"), (1, "R")):
        c.bone("Thigh" + S, P(s * x_leg, hip_y, 0), P(s * x_leg, hip_y * 0.5, -0.1), "Hips")
        c.bone("Shin" + S, P(s * x_leg, hip_y * 0.5, -0.1), P(s * x_leg, 0.3, 0), "Thigh" + S)
        c.bone("Arm" + S, P(s * x_arm, sh_y - 0.2, 0), P(s * x_arm * 1.05, sh_y - 1.8 * arm_len * H, 0), "Chest")
        c.bone("Fore" + S, P(s * x_arm * 1.05, sh_y - 1.8 * arm_len * H, 0), P(s * x_arm * 1.08, sh_y - 3.4 * arm_len * H, -0.3), "Arm" + S)
    return dict(hip_y=hip_y, sh_y=sh_y, x_leg=x_leg, x_arm=x_arm)


def biped_anims(c, stride=22, arm=20, bob=0.12, extra=None, sway=3):

            "ThighL": (-stride * sin(p), 0, 0), "ThighR": (stride * sin(p), 0, 0),

            "ArmL": (arm * sin(p), 0, 4), "ArmR": (-arm * sin(p), 0, -4),

        }
        if extra:
            out.update(extra(t, True))
        return out

    def idle(t):
        p = tau * t
        out = {
            "Root": ((0, 0, 0), (0, 0.05 * sin(2 * p), 0)),

def spine(c, pts, names=None, parent="Root"):
    """chain of bones through pts; returns names"""
    names = names or ["Spine%d" % (i + 1) for i in range(len(pts) - 1)]
    prev = parent
    for i in range(len(pts) - 1):
        c.bone(names[i], pts[i], pts[i + 1], prev)
        prev = names[i]
    return names


def swim_anims(c, chain, fins=(), amp=10, axis="yaw", rate=1, extra=None, hover=0.4):
    """undulating body: each chain bone rotates with a travelling wave. axis yaw (snake/fish) or pitch (whale)"""
    def mk(speed, a_mul):
        def fn(t):
            p = tau * t * rate
            out = {"Root": ((0, 0, 0), (0, hover * sin(p if speed < 1 else 2 * p), 0))}
            for i, n in enumerate(chain):
                ang = amp * a_mul * sin(p * speed - i * 0.9) * (0.4 + 0.6 * i / max(1, len(chain) - 1))
                out[n] = (ang, 0, 0) if axis == "pitch" else (0, ang, 0)
            for i, (n, side) in enumerate(fins):
                out[n] = (0, 0, side * (14 * a_mul) * sin(p * speed * 1.3 + i))
            if extra:
                out.update(extra(t, speed > 1))
            return out
        return fn
    c.animate({"Walk": (24, 1.0, mk(2, 1.3)), "Idle": (40, 4.0, mk(1, 0.8))})


def aura(c, style, color, color2=None, rarity=None):

        c.extra["rarity"] = rarity


def stripes(col, axis="z", period=0.7, width=0.3, wobble=0.25, only=None):
    """painted stripes (per pixel). axis z = bands around the body. only = fn(x,y,z)->bool array"""
    def mask(X, Y, Z):
        U = {"z": Z, "y": Y, "x": X}[axis] + wobble * np.sin(X * 2.2 + Y * 1.3) + 0.12 * np.sin(Y * 5.1)
        m = np.mod(U, period) < width * (0.75 + 0.25 * np.cos(X * 1.7))
        if only is not None:
            m &= only(X, Y, Z)
        return m
    return (col, mask)


def spots(col, size=0.5, density=0.3, seed=1, only=None):
    def mask(X, Y, Z):
        cx, cy, cz = np.floor(X / size), np.floor(Y / size), np.floor(Z / size)
        h = np.sin(cx * 12.9898 + cy * 78.233 + cz * 37.719 + seed) * 43758.5453
        h = h - np.floor(h)
        fx, fy, fz = X / size - cx - 0.5, Y / size - cy - 0.5, Z / size - cz - 0.5
        m = (h < density) & (fx * fx + fy * fy + fz * fz < 0.16)
        if only is not None:
            m &= only(X, Y, Z)
        return m
    return (col, mask)


def band(col, lo, hi, axis="y"):
    """solid painted region between two heights (e.g. a darker back / lighter belly)"""
    def mask(X, Y, Z):
        U = {"z": Z, "y": Y, "x": X}[axis]
        return (U >= lo) & (U <= hi)
    return (col, mask)

"""Detail toolkit for the fierce armoured / royal / elemental creatures (batch 10+).

Everything here is written in GAME space: V(x, y, z) with y up, z back, front = -z. B() converts to
blender. The look is faceted (low segment counts, flat shading) like the owner's reference models:
shard fur clumps, crisp armour plates with trims, glowing angry eyes, open jaws with sharp teeth.
"""
import math, random
from kit import *
import bmesh
import numpy as np
from mathutils import Vector, Matrix, Euler
from mathutils.bvhtree import BVHTree

R = math.radians
GM = G.to_4x4()
TOOTH = C(250, 248, 236)
MOUTH = C(85, 12, 22)
TONGUE = C(215, 70, 95)


def V(x, y=None, z=None):
    if y is None:
        return Vector(x)
    return Vector((x, y, z))


def B(v):
    return P(v[0], v[1], v[2])


def grot(rx=0, ry=0, rz=0):
    return Euler((R(rx), R(ry), R(rz)), "XYZ").to_matrix()


def mirror(v):
    return V(-v.x, v.y, v.z)


def _finish(c, bm, color, bones, **kw):
    bmesh.ops.transform(bm, matrix=GM, verts=bm.verts)     # game -> blender
    bmesh.ops.recalc_face_normals(bm, faces=bm.faces)
    return c._add(bm, color, bones, **kw)


# ------------------------------------------------------------------ basic wrappers (game space)
def E(c, ctr, rad, col, bones, rot=None, seg=(10, 7), **kw):
    return c.ellipsoid(B(ctr), rad, col, bones, rot=euler(*rot) if rot else None, seg=seg, **kw)


def T(c, pts, radii, col, bones, seg=8, sub=2, **kw):
    return c.tube([B(p) for p in pts], radii, col, bones, seg=seg, sub=sub, **kw)


def S(c, base, tip, r, col, bones, bend=None, seg=4, n=3, **kw):
    """faceted spike (pyramid by default)"""
    return c.spike(B(base), B(tip), r, col, bones, bend=B(bend) if bend is not None else None, seg=seg, n=n, **kw)


def BOX(c, ctr, half, col, bones, rot=None, **kw):

# ------------------------------------------------------------------ shard fur
def shard(c, base, tip, width, col, bones, n=None, thick=0.38, mid=0.42, bulge=1.12, lift=0.1, **kw):
    """one faceted fur clump / scale / feather: flat diamond blade lying on the surface (normal n)"""
    d = tip - base
    L = d.length
    dn = d / L
    n = V(0, 1, 0) if n is None else n.copy()
    n = n - dn * n.dot(dn)
    if n.length < 1e-4:
        n = dn.orthogonal()
    n.normalize()
    w = dn.cross(n).normalized()
    bm = bmesh.new()

    def ring(p, s):
        hw, ht = width * 0.5 * s, width * thick * 0.5 * s
        return [bm.verts.new(p + w * hw), bm.verts.new(p + n * ht), bm.verts.new(p - w * hw), bm.verts.new(p - n * ht)]
    r0 = ring(base, 0.8)
    r1 = ring(base + d * mid + n * L * lift * 0.5, bulge)
    tp = bm.verts.new(tip + n * L * lift)
    for i in range(4):
        j = (i + 1) % 4
        bm.faces.new((r0[i], r0[j], r1[j], r1[i]))
        bm.faces.new((r1[i], r1[j], tp))
    bm.faces.new(list(reversed(r0)))
    return _finish(c, bm, col, bones, **kw)


def fibo(n):
    ga = math.pi * (3 - math.sqrt(5))
    out = []
    for i in range(n):
        y = 1 - 2 * (i + 0.5) / n
        r = math.sqrt(max(0, 1 - y * y))
        out.append(V(math.cos(ga * i) * r, y, math.sin(ga * i) * r))
    return out


def fur(c, ctr, rad, col, bones, n=80, length=0.9, width=0.55, flow=V(0, -1, 0.3), where=None, col2=None, mix=0.35,
        seed=1, rot=None, lift=0.45, jitter=0.3, thick=0.38, sink=0.3, glow2=False, wjit=0.2):
    """cover part of an ellipsoid in shard clumps. where(u, p) filters by unit-sphere dir u / point p.
    flow = Vector or fn(p, normal) -> Vector (direction the fur lies)."""
    rng = random.Random(seed)
    M = grot(*rot) if rot else Matrix.Identity(3)
    count = 0
    for u in fibo(n):
        p = ctr + M @ V(u.x * rad[0], u.y * rad[1], u.z * rad[2])
        if where and not where(u, p):
            continue
        nrm = (M @ V(u.x / rad[0], u.y / rad[1], u.z / rad[2])).normalized()
        f = flow(p, nrm) if callable(flow) else flow.copy()
        f = f - nrm * f.dot(nrm)
        if f.length < 1e-3:
            f = nrm.orthogonal()
        f.normalize()
        dirv = (f + nrm * lift).normalized()
        ln = length * (1 + rng.uniform(-jitter, jitter))
        wd = width * (1 + rng.uniform(-wjit, wjit))
        base = p - dirv * ln * 0.18 - nrm * wd * thick * sink
        cc = col2 if (col2 is not None and rng.random() < mix) else col
        shard(c, base, base + dirv * ln, wd, cc, bones, n=nrm, thick=thick, glow=bool(glow2 and cc == col2))
        count += 1
    return count


def fur_line(c, pts, col, bones, length=0.9, width=0.5, out=None, n_per=3, col2=None, seed=2, lift=0.6, thick=0.4, jitter=0.25):
    """row of shards along a polyline (manes, ridges, cheek tufts). out = fn(p) -> direction"""
    rng = random.Random(seed)
    for i in range(len(pts) - 1):
        for k in range(n_per):
            t = (k + rng.uniform(0.1, 0.9)) / n_per
            p = pts[i].lerp(pts[i + 1], t)
            d = out(p) if callable(out) else out
            d = d.normalized()
            ln = length * (1 + rng.uniform(-jitter, jitter))
            seg_dir = (pts[i + 1] - pts[i]).normalized()
            nrm = d.cross(seg_dir).cross(d) if abs(d.dot(seg_dir)) < 0.99 else d.orthogonal()
            cc = col2 if (col2 is not None and rng.random() < 0.4) else col
            shard(c, p - d * ln * 0.15, p + d * ln, width * (1 + rng.uniform(-0.2, 0.2)), cc, bones, n=nrm, thick=thick, lift=0.05)


# ------------------------------------------------------------------ armour
def shell(c, ctr, rad, col, bones, th=(-60, 60), ph=(-30, 60), off=0.06, thick=0.16, seg=(6, 4), rot=None,
          trim=None, inset=0.16, **kw):
    """curved armour plate over part of an ellipsoid. th = azimuth range in degrees (0 = front -z, 90 = +x,
    180 = back), ph = elevation range. trim=colour -> plate in trim colour + raised inset panel in col."""
    M = grot(*rot) if rot else Matrix.Identity(3)
    kw.setdefault("stud", False)          # armour plates are clean hard surfaces

    def surf(t, p, o):
        t, p = R(t), R(p)
        u = V(math.sin(t) * math.cos(p), math.sin(p), -math.cos(t) * math.cos(p))
        pt = V(u.x * rad[0], u.y * rad[1], u.z * rad[2])
        nr = V(u.x / rad[0], u.y / rad[1], u.z / rad[2]).normalized()
        return ctr + M @ (pt + nr * o)

    def slab(t0, t1, p0, p1, o0, o1, color, **k2):
        bm = bmesh.new()
        nu, nv = seg
        A = [[bm.verts.new(surf(t0 + (t1 - t0) * i / nu, p0 + (p1 - p0) * j / nv, o1)) for j in range(nv + 1)] for i in range(nu + 1)]
        Bq = [[bm.verts.new(surf(t0 + (t1 - t0) * i / nu, p0 + (p1 - p0) * j / nv, o0)) for j in range(nv + 1)] for i in range(nu + 1)]
        for i in range(nu):
            for j in range(nv):
                bm.faces.new((A[i][j], A[i + 1][j], A[i + 1][j + 1], A[i][j + 1]))
                bm.faces.new((Bq[i][j], Bq[i][j + 1], Bq[i + 1][j + 1], Bq[i + 1][j]))
        edge = [(i, 0) for i in range(nu)] + [(nu, j) for j in range(nv)] + [(i, nv) for i in range(nu, 0, -1)] + [(0, j) for j in range(nv, 0, -1)]
        for k in range(len(edge)):
            (i1, j1), (i2, j2) = edge[k], edge[(k + 1) % len(edge)]
            bm.faces.new((A[i1][j1], Bq[i1][j1], Bq[i2][j2], A[i2][j2]))
        return _finish(c, bm, color, bones, **k2)

    if trim is None:
        return slab(th[0], th[1], ph[0], ph[1], off, off + thick, col, **kw)
    slab(th[0], th[1], ph[0], ph[1], off, off + thick, trim, **kw)
    dt, dp = (th[1] - th[0]) * inset, (ph[1] - ph[0]) * inset
    return slab(th[0] + dt, th[1] - dt, ph[0] + dp, ph[1] - dp, off + thick * 0.5, off + thick * 1.5, col, **kw)


def cuff(c, a, b, r, col, bones, thick=0.14, seg=8, flare=0.15, spikes=0, spike_col=None, spike_len=0.5, **kw):
    """armour ring / gauntlet / greave around a limb from a to b (game). r = inner radius at the middle"""
    kw.setdefault("stud", False)
    d = b - a
    dn = d.normalized()
    s1 = dn.cross(V(0, 1, 0)) if abs(dn.y) < 0.95 else dn.cross(V(1, 0, 0))
    s1.normalize()
    s2 = dn.cross(s1).normalized()
    bm = bmesh.new()
    prof = [(0.0, r + flare), (0.5, r), (1.0, r + flare)]
    outer, inner = [], []
    for t, rr in prof:
        ctr = a + d * t
        dirs = [s1 * math.cos(k / seg * math.tau) + s2 * math.sin(k / seg * math.tau) for k in range(seg)]
        outer.append([bm.verts.new(ctr + q * (rr + thick)) for q in dirs])
        inner.append([bm.verts.new(ctr + q * rr) for q in dirs])
    for i in range(len(prof) - 1):
        for k in range(seg):
            k2 = (k + 1) % seg
            bm.faces.new((outer[i][k], outer[i][k2], outer[i + 1][k2], outer[i + 1][k]))
            bm.faces.new((inner[i][k], inner[i + 1][k], inner[i + 1][k2], inner[i][k2]))
    for ro, ri in ((outer[0], inner[0]), (outer[-1], inner[-1])):
        for k in range(seg):
            k2 = (k + 1) % seg
            bm.faces.new((ro[k], ri[k], ri[k2], ro[k2]))
    _finish(c, bm, col, bones, **kw)
    if spikes:
        mid = a + d * 0.5
        for k in range(spikes):
            q = s1 * math.cos(k / spikes * math.tau + 0.3) + s2 * math.sin(k / spikes * math.tau + 0.3)
            S(c, mid + q * (r + thick * 0.6), mid + q * (r + thick + spike_len) + dn * spike_len * 0.3, thick * 1.6, spike_col or col, bones, stud=False)


def crown(c, ctr, rad, h, col, bones, n=5, gem=None, tilt=(0, 0, 0), thick=0.18, tall=1.5, emblem=True, points="spike", stud=False):
    """royal crown: ring band + n tall points + front emblem with a glowing gem"""
    M = grot(*tilt)
    cuff(c, ctr, ctr + M @ V(0, h, 0), rad, col, bones, thick=thick, seg=max(10, n * 2), flare=0.06, stud=stud)
    for i in range(n):
        a = i / n * math.tau
        dirv = V(math.sin(a), 0, -math.cos(a))
        tang = V(math.cos(a), 0, math.sin(a))
        ph = h * tall * (1.25 if i == 0 else 1.0)
        base = ctr + M @ (dirv * (rad + thick * 0.5) + V(0, h * 0.85, 0))
        w = rad * math.tau / n * 0.36
        tip = base + M @ (V(0, ph, 0) + dirv * 0.08 * ph)
        SHEET(c, [base - M @ tang * w, base + M @ tang * w, tip], col, bones, thick=thick * 1.2, stud=stud)
        if gem:
            E(c, tip - M @ V(0, ph * 0.15, 0) + M @ dirv * thick * 0.6, (0.09 * rad * 1.5,) * 3, gem, bones, seg=(6, 4), glow=True)
    if emblem:
        f = ctr + M @ V(0, h * 0.55, -(rad + thick))
        BOX(c, f, (h * 0.42, h * 0.42, thick * 0.6), col, bones, rot=(tilt[0], tilt[1], 45 + tilt[2]), stud=stud)
        if gem:
            BOX(c, f + M @ V(0, 0, -thick * 0.45), (h * 0.24, h * 0.24, thick * 0.4), gem, bones, rot=(tilt[0], tilt[1], 45 + tilt[2]), glow=True)


# ------------------------------------------------------------------ face
def _snap(c, point, direction, bones, max_dist=3.0):
    d = direction.normalized()
    start = point + d * max_dist
    best = None
    bset = set(bones if isinstance(bones, (list, tuple)) else [bones])
    for sh in c.shapes:
        if sh["glow"] or not (set(sh["bones"]) & bset):
            continue
        hit = BVHTree.FromBMesh(sh["bm"]).ray_cast(start, -d, max_dist * 2)
        if hit[0] is not None:
            dist = (hit[0] - start).length
            if best is None or dist < best[1]:
                best = (hit[0], dist, hit[1])
    return (best[0], best[2]) if best else (point, d)


def fierce_eye(c, center, size, bones, facing=None, glow=C(255, 45, 45), core=C(255, 150, 120), brow=None,
               slant=20, socket=C(25, 10, 12), brow_len=1.2, brow_lift=0.62, snap=True):
    """glowing angry eye: dark socket, slanted glowing almond (inner corner down), hot core, angry brow"""
    cb, fb = B(center), B(facing if facing is not None else V(0, 0, -1)).normalized()
    if snap:
        cb, nrm = _snap(c, cb, fb, bones)
        fb = (fb * 0.5 + nrm.normalized() * 0.5).normalized()
    f = fb
    inner = Vector((-math.copysign(1, cb.x) if abs(cb.x) > 1e-4 else 1, 0, 0))
    sv = f.cross(Vector((0, 0, 1)))
    if sv.length < 1e-3:
        sv = Vector((1, 0, 0))
    sv.normalize()
    if sv.dot(inner) < 0:
        sv = -sv
    up = Vector((0, 0, 1)) - f * f.z
    up = (up - sv * up.dot(sv)).normalized()

    def axes(deg):
        a = R(deg)
        return sv * math.cos(a) - up * math.sin(a), up * math.cos(a) + sv * math.sin(a)

    def ell(off, rad, col, X, Y, gl):
        bm = bmesh.new()
        bmesh.ops.create_uvsphere(bm, u_segments=10, v_segments=7, radius=1.0)
        Mx = Matrix((X, f, Y)).transposed().to_4x4()
        m = Matrix.Translation(cb + X * off[0] + f * off[1] + Y * off[2]) @ Mx @ Matrix.Diagonal((rad[0], rad[1], rad[2], 1))
        bmesh.ops.transform(bm, matrix=m, verts=bm.verts)
        bmesh.ops.recalc_face_normals(bm, faces=bm.faces)
        c._add(bm, col, bones, glow=gl, stud=False)
    s = size
    ax, bx = axes(slant)
    ell((0, -s * 0.1, 0), (s * 0.95, s * 0.2, s * 0.6), socket, ax, bx, False)
    ell((0, 0.0, 0), (s * 0.74, s * 0.14, s * 0.38), glow, ax, bx, True)
    ell((-s * 0.08, s * 0.04, s * 0.02), (s * 0.2, s * 0.12, s * 0.14), core, ax, bx, True)
    if brow:
        ax2, bx2 = axes(slant * 1.6)
        bm = bmesh.new()
        bmesh.ops.create_cube(bm, size=2.0)
        Mx = Matrix((ax2, f, bx2)).transposed().to_4x4()
        m = Matrix.Translation(cb + bx * s * brow_lift + f * s * 0.08 + ax * s * 0.12) @ Mx @ Matrix.Diagonal((s * 0.78 * brow_len, s * 0.3, s * 0.22, 1))
        bmesh.ops.transform(bm, matrix=m, verts=bm.verts)
        bmesh.ops.recalc_face_normals(bm, faces=bm.faces)
        c._add(bm, brow, bones)

def maw(c, F, w, d, gap, upper, jaw, n=8, tooth=0.24, fangs=1.8, teeth_col=TOOTH, mouth_col=MOUTH, tongue_col=TONGUE,
        lower_teeth=True, tongue=True, inner=True):
    """open mouth with rows of sharp teeth. F = front centre of the mouth opening (game), w = half width,
    d = depth back to the corners, gap = opening height. Upper teeth on `upper` bone, lower on `jaw`."""
    if inner:
        E(c, F + V(0, 0, d * 0.78), (w * 0.95, gap * 0.6, d * 0.85), mouth_col, [upper], stud=False, seg=(10, 7))
    if tongue:
        E(c, F + V(0, -gap * 0.3, d * 0.5), (w * 0.6, gap * 0.17, d * 0.62), tongue_col, [jaw], stud=False, seg=(8, 6))
    rows = [(gap * 0.5, -1, upper, 1.0)] + ([(-gap * 0.5, 1, jaw, 0.8)] if lower_teeth else [])
    for yy, sgn, bone, scale in rows:
        for i in range(n):
            t = -1 + 2 * i / (n - 1)
            x = t * w * 0.9
            z = F.z + d * t * t * 0.95 + 0.04
            canine = abs(abs(t) - 0.62) < 0.15
            ln = tooth * (fangs if canine else 1.0) * scale
            base = V(x, F.y + yy - sgn * 0.04, z)
            tip = base + V(-x * 0.04, sgn * ln, 0)
            S(c, base, tip, tooth * 0.38 * (1.2 if canine else 1), teeth_col, bone, stud=False)


def ram_horn(c, base, s, size, col, bones, turns=0.85, out=0.5, yaw=25, ribs=True, seg=7, r0=None, **kw):
    """curled ram horn starting at `base` (game) on side s: goes up, back, down and forward, drifting outward"""
    M = grot(0, s * yaw, 0)
    pts, radii = [], []
    n = 16
    r0 = r0 or size * 0.42
    for i in range(n + 1):
        k = i / n
        a = k * turns * math.tau
        rr = size * (1 - 0.45 * k)
        off = V(s * out * size * k * 1.6, rr * math.cos(a) - size, rr * math.sin(a))
        pts.append(base + M @ off)
        r = r0 * (1 - 0.82 * k) + 0.04
        radii.append(r * (1.12 if ribs and i % 2 else 1.0))
    return T(c, pts, radii, col, bones, seg=seg, sub=1, **kw)


def swept_horn(c, pts, r, col, bones, seg=6, ribs=False, **kw):
    n = len(pts)
    radii = [r * (1 - 0.88 * i / (n - 1)) + 0.03 for i in range(n)]
    if ribs:
        radii = [rr * (1.1 if i % 2 else 1) for i, rr in enumerate(radii)]
    return T(c, pts, radii, col, bones, seg=seg, sub=2 if not ribs else 1, **kw)

def filigree(col, scale=1.0, width=0.09, only=None):
    """swirling gold trim lines"""
    def mask(X, Y, Z):
        u = np.sin(X * 1.7 * scale + np.sin(Z * 1.3 * scale) * 1.8) + 0.8 * np.sin(Z * 1.1 * scale + np.sin(Y * 1.9 * scale) * 1.5) + np.sin(Y * 1.4 * scale)
        m = np.abs(u) < width * 3
        if only is not None:
            m &= only(X, Y, Z)
        return m
    return (col, mask)


def cracks(col, scale=1.0, width=0.06, only=None):
    """glowing-lava / lightning style crack network (painted)"""
    def mask(X, Y, Z):
        a = np.abs(np.sin(X * 2.1 * scale + np.sin(Y * 1.7 * scale) * 2.2 + Z * 0.7 * scale))
        b = np.abs(np.sin(Z * 2.3 * scale + np.sin(X * 1.3 * scale) * 2.0 - Y * 0.9 * scale))
        m = (a < width * 2.5) | (b < width * 2.0)
        if only is not None:
            m &= only(X, Y, Z)
        return m
    return (col, mask)

def beast(c, sh=4.2, hip=None, L=2.0, W=1.1, neck=(1.0, 1.0), head=1.5, tail=4, tail_len=1.0, tail_lift=0.25):
    """quadruped skeleton compatible with kit.quad_anims (+ a Jaw bone). Chest at z=-L, hips at z=+L.

    hip = hip or sh
    cz, hz = -L, L
    mid = V(0, (hip + sh) / 2, 0)
    c.bone("Root", P(0, 0, 0), P(0, 1, 0))
    c.bone("Hips", B(V(0, hip, hz)), B(mid), "Root")
    c.bone("Chest", B(mid), B(V(0, sh, cz)), "Hips")
    nb = V(0, sh + 0.3, cz - 0.2)
    hb = nb + V(0, neck[1], -neck[0])
    c.bone("Neck", B(nb), B(hb), "Chest")
    c.bone("Head", B(hb), B(hb + V(0, 0.1, -head)), "Neck")
    c.bone("Jaw", B(hb + V(0, -0.35, -0.3)), B(hb + V(0, -0.7, -head)), "Head")
    k = dict(sh=sh, hip=hip, cz=cz, hz=hz, W=W, nb=nb, hb=hb, mid=mid)
    for s, X in ((-1, "L"), (1, "R")):
        sho, elb, paw = V(s * W, sh - 0.2, cz), V(s * W * 1.05, sh * 0.5, cz - 0.15), V(s * W * 1.05, 0.35, cz - 0.1)
        hp_, kn, ft = V(s * W * 0.95, hip - 0.2, hz), V(s * W * 1.02, hip * 0.52, hz - 0.45), V(s * W * 1.05, 0.35, hz + 0.15)
        c.bone("Arm" + X, B(sho), B(elb), "Chest")

        c.bone("Thigh" + X, B(hp_), B(kn), "Hips")
        c.bone("Shin" + X, B(kn), B(ft), "Thigh" + X)

    for i in range(tail):

        prev, t0 = "Tail%d" % (i + 1), t1

    k["tail"] = tp
    return k

        a, b, f, ub, lb = k["sho" + X], k["elb" + X], k["paw" + X], "Arm" + X, "Fore" + X

def theropod_anims(c, tail=5, stride=30, bob=0.18, extra=None, lean=6, head_amp=1.0):
    def walk(t):
        p = tau * t
        out = {
            "Root": ((0, 0, 3 * sin(p)), (0, bob * abs(sin(p)) - bob * 0.5, 0)),
            "Hips": (0, 5 * sin(p), 0),
            "Chest": (2 * sin(2 * p), -4 * sin(p), 0),
            "Neck": (-4 * sin(2 * p + 0.6) * head_amp, 3 * sin(p), 0),
            "Head": (3 * sin(2 * p + 1.2) * head_amp, 0, 0),
            "ThighL": (-stride * sin(p), 0, 0), "ThighR": (stride * sin(p), 0, 0),
            "ShinL": (30 * max(0, sin(p + 0.6)), 0, 0), "ShinR": (30 * max(0, -sin(p + 0.6)), 0, 0),
            "FootL": (-25 * max(0, sin(p + 1.0)), 0, 0), "FootR": (-25 * max(0, -sin(p + 1.0)), 0, 0),
            "ArmL": (10 * sin(p), 0, 0), "ArmR": (-10 * sin(p), 0, 0),
        }
        for i in range(tail):
            out["Tail%d" % (i + 1)] = (2 * sin(2 * p - i), (8 + 2 * i) * sin(p - 0.7 * i - 0.5), 0)
        if extra:
            out.update(extra(t, True))
        return out

    def idle(t):
        p = tau * t
        roar = max(0, sin(math.pi * (t - 0.55) / 0.3)) if 0.55 <= t <= 0.85 else 0.0
        out = {
            "Root": ((0, 0, 0), (0, 0.06 * sin(2 * p), 0)),
            "Chest": (1.5 * sin(2 * p) - 3 * roar, 0, 0),
            "Neck": (2 * sin(p) - 8 * roar, 16 * sin(p) * head_amp * (1 - roar), 0),
            "Head": (3 * sin(2 * p + 1) + 14 * roar, 0, 5 * sin(p) * (1 - roar)),
            "Jaw": (-3 - 26 * roar, 0, 0),
            "ArmL": (6 * sin(2 * p), 0, 0), "ArmR": (6 * sin(2 * p + 1), 0, 0),
        }
        for i in range(tail):

def qmul_vec(q, v):
    w, x, y, z = q
    # rotate v by quaternion
    ux, uy, uz = x, y, z
    vx, vy, vz = v
    cx = uy * vz - uz * vy; cy = uz * vx - ux * vz; cz = ux * vy - uy * vx
    cx2 = uy * cz - uz * cy; cy2 = uz * cx - ux * cz; cz2 = ux * cy - uy * cx
    return (vx + 2 * (w * cx + cx2), vy + 2 * (w * cy + cy2), vz + 2 * (w * cz + cz2))


def num(x, d=3):
    s = ("%." + str(d) + "f") % x
    s = s.rstrip("0").rstrip(".") if "." in s else s
    return "0" if s in ("-0", "") else s


def convert(path):
    d = json.load(open(path))
    bones = d["bones"]
    names = [b["name"] for b in bones]
    lines = ["-- generated from Blender (" + d["name"] + "). Rotations per bone per frame (w,x,y,z, game axes)",

             "return {", '\tName = "%s", Height = %s, Body = "%s",' % (d["name"], num(d["height"]), d.get("body", d["name"] + "_Body"))]
    lines.append("\tBones = {" + ", ".join('{"%s", %s, Vector3.new(%s)}' % (b["name"], ('"%s"' % b["parent"]) if b["parent"] else "nil", ", ".join(num(v) for v in b["head"])) for b in bones) + "},")
    glow = d.get("glow", {})
    if isinstance(glow, dict):
        lines.append("\tGlow = {" + ", ".join('["%s"] = "%s"' % (k, v) for k, v in glow.items()) + "},")
    ex = d.get("extra", {})

                w, x, y, z, tx, ty, tz = fr[n]

        lines.append("\t\t\tQ = {" + ",".join(num(v, 3) for v in q) + "},")
        lines.append("\t\t\tOff = {" + ",".join(num(v, 3) for v in off) + "}},")
    lines.append("\t},")
    lines.append("}")
    src = "\n".join(lines)
    out = os.path.join(OUT, d["name"] + ".lua")
    open(out, "w").write(src)
    return d["name"], len(src)

"""Cartoon UI icons in the style of Kobe's icon set (thick dark outline, bevelled 3D edge, glossy gradient,
diagonal shine streaks). Outputs 512px PNGs."""
import cairosvg, os
OUT = os.path.dirname(os.path.abspath(__file__))
INK = "#160c07"
_id = [0]


def uid(p):
    _id[0] += 1
    return f"{p}{_id[0]}"


def grad(c1, c2, x1=0, y1=0, x2=0, y2=1):
    g = uid("g")
    return g, f'<linearGradient id="{g}" x1="{x1}" y1="{y1}" x2="{x2}" y2="{y2}"><stop offset="0" stop-color="{c1}"/><stop offset="1" stop-color="{c2}"/></linearGradient>'


def piece(d, top, bottom, side, depth=12, outline=13, shine=True, transform=""):
    """a bevelled flat shape: dark side extruded down by `depth`, glossy top face, outline around both"""
    defs, body = [], []
    g, gd = grad(top, bottom)
    defs.append(gd)
    clip = uid("c")
    defs.append(f'<clipPath id="{clip}"><path d="{d}"/></clipPath>')
    t = f' transform="{transform}"' if transform else ""
    body.append(f'<g{t}>')
    # outline around top + side
    body.append(f'<path d="{d}" transform="translate(0,{depth})" fill="{INK}" stroke="{INK}" stroke-width="{outline * 2}" stroke-linejoin="round"/>')
    body.append(f'<path d="{d}" fill="{INK}" stroke="{INK}" stroke-width="{outline * 2}" stroke-linejoin="round"/>')
    body.append(f'<path d="{d}" transform="translate(0,{depth})" fill="{side}"/>')
    body.append(f'<path d="{d}" fill="url(#{g})"/>')
    if shine:
        body.append(f'<g clip-path="url(#{clip})" opacity="0.38">'
                    f'<path d="M-40 150 L150 -40 L185 -40 L-5 150 Z" fill="#fff"/>'
                    f'<path d="M20 230 L240 10 L262 10 L42 230 Z" fill="#fff"/>'
                    f'<path d="M120 300 L330 90 L345 90 L135 300 Z" fill="#fff"/></g>')
        # light rim on the top-left edge of the face
        body.append(f'<path d="{d}" fill="none" stroke="#ffffff" stroke-opacity="0.45" stroke-width="5" clip-path="url(#{clip})" transform="translate(3,4)"/>')
    body.append('</g>')
    return defs, body


def svg(parts, name):
    defs, body = [], []
    for d, b in parts:
        defs += d; body += b
    s = f'<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 256 256" width="512" height="512"><defs>{"".join(defs)}</defs>{"".join(body)}</svg>'
    open(os.path.join(OUT, name + ".svg"), "w").write(s)
    cairosvg.svg2png(bytestring=s.encode(), write_to=os.path.join(OUT, name + ".png"), output_width=512, output_height=512)

"""Generate seamless stud tiles that match the box icons (square inset studs).
Output is black/white on transparent, so it overlays any coloured panel in Roblox."""
import sys
from PIL import Image, ImageDraw, ImageFilter

OUT = sys.argv[1]
S = 4  # supersample factor


def rounded(draw, box, r, fill):
    draw.rounded_rectangle(box, radius=r, fill=fill)


def square_stud(tile=128, stud=86, radius=10, edge=9):
    T, st, r, e = tile * S, stud * S, radius * S, edge * S
    o = (T - st) // 2
    img = Image.new("RGBA", (T, T), (0, 0, 0, 0))

    # soft dark inset body (slightly darker than the panel)
    body = Image.new("RGBA", (T, T), (0, 0, 0, 0))
    rounded(ImageDraw.Draw(body), (o, o, o + st, o + st), r, (0, 0, 0, 34))
    img.alpha_composite(body)

    # dark shading on the top + left inner edges (light comes from bottom-right, like the icons)
    shade = Image.new("L", (T, T), 0)
    d = ImageDraw.Draw(shade)
    rounded(d, (o, o, o + st, o + st), r, 255)
    rounded(d, (o + e, o + e, o + st + e, o + st + e), r, 0)
    dark = Image.new("RGBA", (T, T), (0, 0, 0, 120))
    dark.putalpha(shade.point(lambda v: v * 120 // 255))
    img.alpha_composite(dark)

    # light rim on the bottom + right inner edges
    rim = Image.new("L", (T, T), 0)
    d = ImageDraw.Draw(rim)
    rounded(d, (o, o, o + st, o + st), r, 255)
    rounded(d, (o - e // 2, o - e // 2, o + st - e // 2, o + st - e // 2), r, 0)
    light = Image.new("RGBA", (T, T), (255, 255, 255, 0))
    light.putalpha(rim.point(lambda v: v * 95 // 255))
    img.alpha_composite(light)

    img = img.filter(ImageFilter.GaussianBlur(S * 0.6))
    return img.resize((tile, tile), Image.LANCZOS)


def round_stud(tile=128, stud=78, edge=9):
    T, st, e = tile * S, stud * S, edge * S
    o = (T - st) // 2
    img = Image.new("RGBA", (T, T), (0, 0, 0, 0))
    # drop shadow down-right
    sh = Image.new("RGBA", (T, T), (0, 0, 0, 0))
    ImageDraw.Draw(sh).ellipse((o + e, o + e, o + st + e, o + st + e), fill=(0, 0, 0, 90))
    img.alpha_composite(sh.filter(ImageFilter.GaussianBlur(S * 2)))
    # stud top
    ImageDraw.Draw(img).ellipse((o, o, o + st, o + st), fill=(255, 255, 255, 40))
    # highlight crescent top-left
    hl = Image.new("L", (T, T), 0)
    d = ImageDraw.Draw(hl)
    d.ellipse((o, o, o + st, o + st), fill=255)
    d.ellipse((o + e, o + e, o + st + e, o + st + e), fill=0)
    white = Image.new("RGBA", (T, T), (255, 255, 255, 0))
    white.putalpha(hl.point(lambda v: v * 150 // 255))
    img.alpha_composite(white)
    return img.filter(ImageFilter.GaussianBlur(S * 0.6)).resize((tile, tile), Image.LANCZOS)


def preview(tiles, path):
    colours = [(235, 40, 60), (230, 40, 160), (70, 60, 230), (60, 200, 70), (255, 190, 30)]
    W = 260
    out = Image.new("RGBA", (W * len(colours), W * len(tiles)), (40, 40, 40, 255))
    for row, t in enumerate(tiles):
        small = t.resize((40, 40), Image.LANCZOS)
        for col, c in enumerate(colours):
            panel = Image.new("RGBA", (W - 10, W - 10), c + (255,))
            for y in range(0, W, 40):
                for x in range(0, W, 40):
                    panel.alpha_composite(small, (x, y)) if x + 40 <= W - 10 and y + 40 <= W - 10 else None
            out.alpha_composite(panel, (col * W + 5, row * W + 5))
    out.save(path)

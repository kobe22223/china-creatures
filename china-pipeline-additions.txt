##### FILE: kit.py ADDITION (cute_quad) #####
def cute_quad(c, col, belly=None, paw=None, body=(1.45, 1.2, 1.75), leg_h=1.35, leg_r=0.42, head_r=1.3,
              head_up=0.55, neck_len=0.35, tail=1, paint=None, spread=0.55):
    """CUTE PET body plan (Kobe's proportion rules): round barrel body, SHORT stubby legs (leg_h about 1/4 of
    total height), big head (about 40-45% of height) sitting forward and up on the body, no visible neck.
    Same bone names as quad(), so quad_anims works. Returns key points (blender vectors) and sizes."""
    rx, ry, rz = body
    by = leg_h + ry * 0.75                     # body centre height
    sz, hz = -rz * 0.55, rz * 0.55            # shoulder / hip z
    lx = rx * spread                           # leg x
    hc = P(0, by + ry * head_up, -rz - head_r * 0.45)   # head centre
    c.bone("Root", P(0, 0, 0), P(0, 1, 0))
    c.bone("Hips", P(0, by, hz), P(0, by, 0), "Root")
    c.bone("Chest", P(0, by, 0), P(0, by + 0.2, sz), "Hips")
    nb = P(0, by + ry * 0.35, -rz * 0.75)
    c.bone("Neck", nb, nb + P(0, neck_len, -neck_len), "Chest")
    c.bone("Head", nb + P(0, neck_len, -neck_len), hc + P(0, 0, -head_r), "Neck")
    top_y = by - ry * 0.45
    for s, S in ((-1, "L"), (1, "R")):
        c.bone("Arm" + S, P(s * lx, top_y, sz), P(s * lx, leg_h * 0.5, sz), "Chest")
        c.bone("Fore" + S, P(s * lx, leg_h * 0.5, sz), P(s * lx, 0.2, sz - 0.05), "Arm" + S)
        c.bone("Thigh" + S, P(s * lx, top_y, hz), P(s * lx, leg_h * 0.5, hz), "Hips")
        c.bone("Shin" + S, P(s * lx, leg_h * 0.5, hz), P(s * lx, 0.2, hz + 0.05), "Thigh" + S)
    t0, prev = P(0, by + ry * 0.2, rz * 0.95), "Hips"
    for i in range(tail):
        t1 = t0 + P(0, 0.35, 0.6)
        c.bone("Tail%d" % (i + 1), t0, t1, prev)
        prev, t0 = "Tail%d" % (i + 1), t1
    belly = belly or col
    pawc = paw or belly
    c.ellipsoid(P(0, by, 0), (rx, ry, rz), col, ["Hips", "Chest"], seg=(24, 16), paint=paint)
    c.ellipsoid(P(0, by - ry * 0.45, -rz * 0.05), (rx * 0.72, ry * 0.5, rz * 0.8), belly, ["Hips", "Chest"], seg=(18, 10))
    for s, S in ((-1, "L"), (1, "R")):
        for z, up, lo in ((sz, "Arm" + S, "Fore" + S), (hz, "Thigh" + S, "Shin" + S)):
            # haunch: rounded mass sunk into the body so the leg grows out of it (no seam, no gap)
            c.ellipsoid(P(s * lx, by - ry * 0.5, z), (leg_r * 1.4, ry * 0.55, leg_r * 1.5), col, up, seg=(16, 12), paint=paint)
            # short stub leg + rounded paw
            c.ellipsoid(P(s * lx, leg_h * 0.5 + 0.1, z), (leg_r, leg_h * 0.55, leg_r * 1.05), col, lo, seg=(16, 10), paint=paint)
            c.ellipsoid(P(s * lx, 0.27, z - leg_r * 0.2), (leg_r * 1.1, 0.28, leg_r * 1.3), pawc, lo, seg=(14, 8))
    return dict(by=by, rx=rx, ry=ry, rz=rz, sz=sz, hz=hz, lx=lx, head=hc, head_r=head_r, leg_h=leg_h)

"""LUCKY PIGLET (Common, Rice Paddy Village): round pink piglet on short stubby legs, big head, floppy ears,
red lucky bib with a gold coin, curly tail. Cute: plain flat black eyes."""
from kit import *

PINK, PINK2, BELLY, SNOUT, RED, GOLD, HOOF = C(255, 176, 186), C(240, 140, 158), C(255, 205, 212), C(255, 158, 172), C(214, 32, 40), C(255, 200, 60), C(140, 85, 92)


def build():
    c = Creature("LuckyPiglet", height=6)
    k = cute_quad(c, PINK, belly=BELLY, paw=HOOF, body=(1.5, 1.25, 1.8), leg_h=1.2, leg_r=0.45, head_r=1.3, head_up=0.5, tail=2)
    h, r = k["head"], k["head_r"]
    c.ellipsoid(h, (r * 1.05, r * 0.95, r * 0.95), PINK, "Head", seg=(24, 16))
    # snout: short flat disc on the front of the face
    c.tube([h + P(0, -0.15, -r * 0.8), h + P(0, -0.2, -r * 1.12)], [0.55, 0.52], SNOUT, ["Head"], seg=18, sub=1)
    for s in (-1, 1):
        c.ellipsoid(h + P(s * 0.2, -0.2, -r * 1.2), (0.09, 0.15, 0.04), C(150, 70, 85), "Head", stud=False, smooth=True)
        c.eye(h + P(s * 0.5, 0.3, -r * 0.82), 0.24, P(s * 0.45, 0.05, -1), "Head")
        c.ellipsoid(h + P(s * 0.85, -0.25, -r * 0.62), (0.24, 0.14, 0.1), C(255, 130, 150), "Head", stud=False)
        # floppy ears: start on top of the head, fold forward over the brow
        e0 = h + P(s * 0.55, r * 0.82, -0.3)
        c.leaf(e0, P(s * 0.75, -0.55, -0.55), P(0, 0, 1), 1.05, 0.85, PINK2, "Head", thick=0.22, shape="blade", curl=-0.25)
    # red lucky bib + gold coin, sitting on the chest under the chin
    bib = P(0, k["by"] + 0.15, -k["rz"] * 0.88)
    c.ellipsoid(bib, (1.0, 0.75, 0.3), RED, "Chest", seg=(16, 10))
    coin = bib + P(0, -0.1, -0.3)
    c.torus(coin, 0.3, 0.08, GOLD, "Chest", rot=euler(90, 0, 0), seg=(16, 6))
    c.box(coin, (0.1, 0.1, 0.07), GOLD, "Chest", stud=False)
    # curly tail
    t = P(0, k["by"] + 0.3, k["rz"] * 0.95)
    pts = [t, t + P(0.3, 0.3, 0.25), t + P(0.0, 0.55, 0.15), t + P(-0.2, 0.35, 0.0), t + P(0.05, 0.25, -0.05)]
    c.tube(pts, [0.15, 0.14, 0.12, 0.1, 0.07], PINK2, ["Tail1", "Tail2"], seg=8)
    aura(c, "sparkle", (255, 200, 60), (255, 120, 120), "Common")
    c.extra["speed"] = 1.0
    return c


def anims(c):
    def extra(t, walking):
        p = tau * t
        return {"Tail1": (0, 0, 18 * sin(4 * p)), "Head": (4 * sin(2 * p), 0, 6 * sin(p))}
    quad_anims(c, tail=2, stride=26, bob=0.12, extra=extra)
